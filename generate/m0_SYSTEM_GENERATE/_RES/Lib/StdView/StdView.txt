	"Html"
		function "MdTokenVertexesToHTMLString_Transform" (@Event "event", @VertexType "from", @VertexType "meta", @VertexType "to")
			variable "html" @String
			
			html = ""
			
			for edge "e" in from\
				test e\Meta
					case "Text"
						html += e\To
						
					case "HardBreak"
						html += "<BR>"
						
					case "ParagraphBreak"
						html += "<BR><BR>"
						
					case "HorizontalRule"
						html += "<HR>"
						
					case "Header1Start"
						html += "<H1>"
											
					case "Header2Start"
						html += "<H2>"
						
					case "Header3Start"
						html += "<H3>"
						
					case "Header4Start"
						html += "<H4>"
						
					case "Header5Start"
						html += "<H5>"
						
					case "Header6Start"
						html += "<H6>"
						
					case "Header1End"
						html += "</H1>"
											
					case "Header2End"
						html += "</H2>"
						
					case "Header3End"
						html += "</H3>"
						
					case "Header4End"
						html += "</H4>"
						
					case "Header5End"
						html += "</H5>"
						
					case "Header6End"
						html += "</H6>"
						
					case "BoldStart"
						html += "<B>"
						
					case "BoldEnd"
						html += "</B>"
						
					case "ItalicStart"
						html += "<I>"
						
					case "ItalicEnd"
						html += "</I>"
						
					case "CodeBlockStart"
						html += "<div style=code>"
						
					case "CodeBlockEnd"
						html += "</div>"
						
					case "InlineCodeStart"
						html += "<div style=inlinecode>"
						
					case "InlineCodeEnd"
						html += "</div>"
						
					case "LinkStart"
						html += ""
						
					case "LinkText"
						html += ""
						
					case "LinkUrl"
						html += ""
						
					case "LinkEnd"
						html += ""
						
					case "ImageStart"
						html += ""
						
					case "ImageUrl"
						html += ""
						
					case "ImageEnd"
						html += ""
						
					case "BlockquoteStart"
						html += "[bqBEG]"
						
					case "BlockquoteEnd"
						html += "[bqEND]"
						
					case "ListItemsStart"
						html += "<ul>"
						
					case "ListItemStart"
						html += "<li>"
						
					case "ListItemsEnd"
						html += "</ul>"
					
					case "OrderedListItemsStart"
						html += "<ol>"
						
					case "OrderedListItemStart"
						html += "<li>"
						
					case "OrderedListItemsEnd"
						html += "</ol>"
						
					case "TableStart"
						html += "<table>"
						
					case "HeaderBegin"
						html += "<tr>"
						
					case "HeaderEnd"
						html += "</tr>"
						
					case "TableEnd"
						html += "</table>"
						
					case "RowBegin"
						html += "<tr>"
						
					case "RowEnd"
						html += "</tr>"
						
					case "HeaderColumnBegin"
						html += "<td>"
						
					case "HeaderColumnEnd"
						html += "</td>"
						
					case "CellBegin"
						html += "<td>"
						
					case "CellEnd"
						html += "</td>"
						
					case "AlignLeft"
						html += "<div align=left>"
						
					case "AlignCenter"
						html += "<div align=center>"
						
					case "AlignRight"
						html += "<div align=right>"
						
				
			to <- @@Concatenate[html]
